Undecidability by EPF axiom

I wrote code that proves undecidability of partial computable functions, only using EPF axiom.

I reffered to work of Dominik Kirst, 


import Godel.Basic
import Mathlib.Data.Real.Basic


set_option linter.style.whitespace false
set_option linter.style.emptyLine false

-- EPF

def IsUniversal {Y : Type} (Θ : ℕ → (ℕ → Option Y)) :=
∀ f :  ℕ → Option Y , ∃ c : ℕ,  ∀ x y , ((Θ c x) = some y ↔ f x = some y)


axiom EPF (Y : Type) : ∃ Θ : ℕ → (ℕ → Option Y), IsUniversal Θ

variable (Θ : ℕ → (ℕ → Option Bool))

def self_halting : ℕ → Prop := fun x => (∃ y , Θ x x = some y )

lemma self_halting_diverge
  (Θ : ℕ → (ℕ → Option Bool))
  (f : ℕ → Option Bool) :
  (IsUniversal Θ) →
  (∀ c, f c = some true ↔ self_halting Θ c) →
  (∃ c, ∀ y, ¬ f c = some y) := by
  intro h h'
  unfold IsUniversal at h

  let g : ℕ → Option Bool := fun x =>
    match f x with
    | some true => none
    | some false      => some true
    | none => none
  have h'' := h g
  rcases h'' with ⟨c', h_c'⟩
  use c'
  intro y contra
  have h_c'' := h_c' c' y
  cases y
  · -- Case 1: y = false (contra : f c' = some false)
    have h_g_none : g c' = true := by
      dsimp [g]
      rewrite [contra]
      rfl
    have := h' c'
    rw [h_g_none] at h_c''
    unfold self_halting at this
    have non_true : ¬ f c' = some true := by
      intro hh
      rw [contra] at hh
      contradiction
    rw [this] at non_true
    have h_iff_true := h_c' c' true
    have h_theta_true : Θ c' c' = some true := h_iff_true.mpr h_g_none
    have h_halt : ∃ y, Θ c' c' = some y := ⟨true, h_theta_true⟩
    exact non_true h_halt
  · -- Case 2: y = true (contra : f c' = some true)
    have h_g_none : g c' = none := by
      dsimp [g]
      rewrite [contra]
      rfl
    have := h' c'
    rw [h_g_none] at h_c''
    rw [this] at contra
    unfold self_halting at contra
    rcases contra with ⟨y0, h_theta⟩
    have h_iff := h_c' c' y0
    have h_g_some : g c' = some y0 := h_iff.mp h_theta
    rw [h_g_none] at h_g_some
    nomatch h_g_some
